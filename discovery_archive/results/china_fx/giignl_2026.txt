  const PUBLIC_BASE = 'https://giignl-documents.s3.fr-par.scw.cloud/'

  function getMemberstackToken() {
    try {
      const token = window.$memberstackDom.getMemberCookie()
      return token || null
    } catch (_) {
      return null
    }
  }

  async function handleDownload(btn) {
    const fileKey = btn.getAttribute('data-pdf-key')
    if (!fileKey) return

    // Nom affiché optionnel (ex: data-filename="GIIGNL Annual Report 2026.pdf")
    const filename = btn.getAttribute('data-filename') || undefined

    // État chargement — on préserve le HTML interne (icône incluse) au lieu de l'écraser
    const originalHTML = btn.innerHTML
    btn.classList.add('giignl-loading')
    btn.disabled = true

    try {
      const token = await getMemberstackToken()
      if (!token) {
        alert('Vous devez être connecté pour télécharger ce document.')
        return
      }

      const res = await fetch(ENDPOINT, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(filename ? { fileKey, filename } : { fileKey }),
      })

      if (res.status === 401 || res.status === 403) {
        alert('Accès non autorisé. Veuillez vous reconnecter.')
        return
      }

      if (!res.ok) {
        alert('Une erreur est survenue. Veuillez réessayer.')
        return
      }

      const { url } = await res.json()
      window.open(url, '_blank')

    } catch (err) {
      alert('Une erreur est survenue. Veuillez réessayer.')
      console.error('[GIIGNL] Erreur download:', err)
    } finally {
      btn.innerHTML = originalHTML
      btn.classList.remove('giignl-loading')
      btn.disabled = false
    }

  // sur chaque bouton disparaîtrait avec eux. La délégation survit à ces re-rendus.
  document.addEventListener('click', function (e) {
    const btn = e.target.closest('[data-pdf-key]')
    if (!btn) return
    e.preventDefault()
    handleDownload(btn)
  })

    const a = e.target.closest('[data-public-key]')
